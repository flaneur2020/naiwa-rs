+        /// Queue device-to-device copies: `dst` gets `src`'s first `dst.len` bytes, once the *source's* readers are
+        /// done and the *destination's* are too.
+        ///
+        /// This is how a staged expert is promoted into the cache: its bytes already crossed the link, so
+        /// admitting it costs a d2d copy instead of a second H2D. Both fences matter - the source ring slot is
+        /// being read by the step's kernels, and the destination slot by whatever still holds it.
+        pub fn queue_d2d(
+            &self,
+            src_fence: &ReadFence,
+            dst_fence: &ReadFence,
+            parts: &[(&SlotMemory, &SlotMemory)],
+        ) -> Result<()> {
+            self.stream
+                .context()
+                .bind_to_thread()
+                .map_err(Error::wrap)?;
+            for fence in [src_fence, dst_fence] {
+                if let Some(event) = fence.event.lock().unwrap().as_ref() {
+                    self.stream.wait(event).map_err(Error::wrap)?;
+                }
+            }
+            for (dst, src) in parts {
+                if src.len < dst.len {
+                    candle_core::bail!(
+                        "promotion of {} bytes into a {}-byte slot",
+                        src.len,
+                        dst.len
+                    );
+                }
+                // SAFETY: both are live allocations, `dst.len` bytes are being written and they are a prefix of
+                // the source; the caller keeps the source alive and orders the readers after this stream.
+                unsafe {
+                    result::memcpy_dtod_async(dst.ptr, src.ptr, dst.len, self.stream.cu_stream())
+                }
+                .map_err(Error::wrap)?;
+            }
+            Ok(())
+        }

+    /// Promote the experts staged last step into the cache, up to `NAIWA_PROMOTE_MAX` per layer per step.
+    ///
+    /// A staged expert has **already crossed the link**, so admitting it is a device-to-device copy rather than a
+    /// second H2D - it converts the next step's most expensive miss (the CPU path) into a hit for the price of a
+    /// copy. Polluting the cache is the risk, hence the cap and the order (empty slots first, LRU victims after).
+    ///
+    /// Called at the *start* of a step, before `stage_misses` queues this step's copies: the caller records the
+    /// layer's completion event after both, so the existing `wait_staging` orders the readers - and it also means
+    /// the pending `Arc`s are dropped before the ring re-uses those slots (`wait_for_readers` waits on the
+    /// refcount, which a pending clone would hold).
+    pub(crate) fn promote_pending(&self, cs: &Arc<CopyStream>) -> candle_core::Result<usize> {

+        let pending = std::mem::take(&mut *self.promote_pending.lock().unwrap());

+        }

+            // Reserve: an empty slot, else the least-recently-used one.
+            let slot = {
